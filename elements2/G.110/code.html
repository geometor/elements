III.31

In a circle the angle in the semicircle is right, that in a greater segment less than a right angle, and that in a less segment greater than a right angle; and further the angle of the greater segment is greater than a right angle, and the angle of the less segment less than a right angle.

        from geometor.model import Model

            D = model.set_point(76.81, -45.03, classes=["given"], ID="D")

            # Circle ABCD
            model.construct_circle(E, A)

            model.set_segment(A, D)
            model.set_segment(D, C)
            model.set_segment(A, E)

            model.set_section([B, A, F])

            # Polygons

            model.set_polygon([A, B, C])
            model.set_polygon([A, B, E])

            model.set_polygon([A, D, C])

            model.set_wedge(C, A, A, E)
            model.set_wedge(A, C, C, E)
            model.set_wedge(A, F, F, C)
            model.set_wedge(E, A, A, C)
            model.set_wedge(E, A, A, B)

            return model
