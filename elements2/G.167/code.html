VI.5

If two triangles have their sides proportional, the triangles is equiangular and will have those angles equal which the corresponding sides subtend.

g1.py

"""Programmatic construction for Book VI, Proposition 5 (VI.5 - g1).

        Given triangle ABC in Proposition 5.
        """

        from __future__ import annotations

        from geometor.model import Model


        def construct_g1() -> Model:
            """Construct the figure for 0209.g1 (VI.5) using geometor.model."""
            model = Model("VI.5.g1")

            # 1. Points
            A = model.set_point(131, -49, classes=["given"], ID="A")
            B = model.set_point(26, -151, classes=["given"], ID="B")
            C = model.set_point(103, -136, classes=["given"], ID="C")

            # 3. Segments
            model.set_segment(A, B)
            model.set_segment(B, C)
            model.set_segment(C, A)

            # 5. Polygons (Triangle)
            model.set_polygon([A, B, C])

            # 6. Angles (Wedges)
            model.set_wedge(A, B, B, C)
            model.set_wedge(B, A, A, C)
            model.set_wedge(C, A, A, B)

            return model


        if __name__ == "__main__":
            construct_g1()

"""Programmatic construction for Book VI, Proposition 5 (VI.5 - g2).

        Triangles DEF and GEF constructed on common side EF in Proposition 5.
        """

        from __future__ import annotations

        from geometor.model import Model


        def construct_g2() -> Model:
            """Construct the figure for 0209.g2 (VI.5) using geometor.model."""
            model = Model("VI.5.g2")

            # 1. Points
            D = model.set_point(84, -29, classes=["given"], ID="D")
            E = model.set_point(27, -83, classes=["given"], ID="E")
            F = model.set_point(69, -75, classes=["given"], ID="F")
            G = model.set_point(105, -106, ID="G")

            # 3. Segments
            model.set_segment(D, E)
            model.set_segment(E, F)
            model.set_segment(F, D)
            model.set_segment(G, E)
            model.set_segment(G, F)

            # 5. Polygons (Triangles)
            model.set_polygon([D, E, F])
            model.set_polygon([G, E, F])

            # 6. Angles (Wedges)
            model.set_wedge(D, E, E, F)
            model.set_wedge(E, D, D, F)
            model.set_wedge(F, D, D, E)
            model.set_wedge(E, F, F, G)
            model.set_wedge(F, E, E, G)
            model.set_wedge(G, E, E, F)

            return model


        if __name__ == "__main__":
            construct_g2()
