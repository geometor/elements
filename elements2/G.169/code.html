G.169 — Code¶

VI.7

If two triangles have one angle equal to one angle, the sides about other angles proportional, and the remaining angles either both less or both not less than a right angle, the triangles is equiangular and will have those angles equal, the sides about which are proportional.

g1.py

"""Programmatic construction for Book VI, Proposition 7 (VI.7 - g1).

        Case 1 of Proposition 7: two triangles with one angle equal, sides about other angles proportional, and remaining angles both less than a right angle.
        """

        from __future__ import annotations

        from geometor.model import Model


        def construct_g1() -> Model:
            """Construct the figure for 0212.g1 (VI.7) using geometor.model."""
            model = Model("VI.7.g1")

            # 1. Points
            A = model.set_point(62, -29, classes=["given"], ID="A")
            B = model.set_point(28, -156, classes=["given"], ID="B")
            C = model.set_point(147, -174, classes=["given"], ID="C")
            G = model.set_point(137.65, -158.05, ID="G")
            D = model.set_point(194, -26, classes=["given"], ID="D")
            E = model.set_point(177, -94, classes=["given"], ID="E")
            F = model.set_point(240, -105, classes=["given"], ID="F")

            # 3. Segments
            model.set_segment(A, B)
            model.set_segment(B, C)
            model.set_segment(A, G)
            model.set_segment(G, C)
            model.set_segment(A, C)
            model.set_segment(B, G)
            model.set_segment(D, E)
            model.set_segment(E, F)
            model.set_segment(F, D)

            # 4. Collinear Sections
            model.set_section([A, G, C])

            # 5. Polygons (Triangles)
            model.set_polygon([A, B, C])
            model.set_polygon([A, B, G])
            model.set_polygon([B, G, C])
            model.set_polygon([D, E, F])

            # 6. Angles (Wedges)
            model.set_wedge(A, B, B, C)
            model.set_wedge(B, A, A, C)
            model.set_wedge(B, A, A, G)
            model.set_wedge(C, A, A, B)
            model.set_wedge(G, A, A, B)
            model.set_wedge(G, B, B, C)
            model.set_wedge(D, E, E, F)
            model.set_wedge(E, D, D, F)
            model.set_wedge(F, D, D, E)

            return model


        if __name__ == "__main__":
            construct_g1()

"""Programmatic construction for Book VI, Proposition 7, Case 2 (VI.7 - g2).

        If two triangles have one angle equal to one angle, the sides about other
        angles proportional, and the remaining angles both not less than a right angle,
        the triangles will be equiangular.
        """

        from __future__ import annotations

        from geometor.model import Model


        def construct_g2() -> Model:
            """Construct the figure for 0213.g1 (VI.7 Case 2) using geometor.model."""
            model = Model("VI.7.g2")

            # 1. Points
            A = model.set_point(83.0, -29.0, classes=["given"], ID="A")
            B = model.set_point(23.0, -160.0, classes=["given"], ID="B")
            C = model.set_point(108.0, -142.0, classes=["given"], ID="C")
            G = model.set_point(104.5, -126.18, ID="G")
            D = model.set_point(188.0, -56.0, classes=["given"], ID="D")
            E = model.set_point(155.0, -130.0, classes=["given"], ID="E")
            F = model.set_point(201.0, -121.0, classes=["given"], ID="F")

            # 2. Segments
            model.set_segment(A, B)
            model.set_segment(B, C)
            model.set_segment(A, C)
            model.set_segment(A, G)
            model.set_segment(G, C)
            model.set_segment(B, G)

            model.set_segment(D, E)
            model.set_segment(E, F)
            model.set_segment(D, F)

            # 3. Collinear Sections
            model.set_section([A, G, C])

            # 4. Polygons
            model.set_polygon([A, B, C])
            model.set_polygon([A, B, G])
            model.set_polygon([B, G, C])
            model.set_polygon([D, E, F])

            # 5. Angles / Wedges
            model.set_wedge(A, B, B, C)
            model.set_wedge(B, A, A, C)
            model.set_wedge(B, A, A, G)
            model.set_wedge(C, B, B, A)
            model.set_wedge(G, B, B, A)
            model.set_wedge(G, B, B, C)
            model.set_wedge(D, E, E, F)
            model.set_wedge(E, D, D, F)
            model.set_wedge(F, E, E, D)

            return model


        if __name__ == "__main__":
            construct_g2()
